```python
"""円形3分割スタンプPNG生成。

レイアウト：上=名前 / 中=日付 / 下=所属。背景透明、朱色固定。
"""
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

VERMILLION = (200, 16, 46, 255)  # #C8102E
DPI = 600
MM_TO_INCH = 1 / 25.4
PT_TO_PX = DPI / 72


def _mm_to_px(mm: float) -> int:
    return int(round(mm * MM_TO_INCH * DPI))


def _fit_font(text: str, max_width: int, max_height: int, font_path: Path) -> ImageFont.FreeTypeFont:
    """テキストが max_width × max_height に収まる最大フォントサイズを返す。"""
    size = max_height
    while size > 4:
        font = ImageFont.truetype(str(font_path), size)
        bbox = font.getbbox(text)
        w = bbox[2] - bbox[0]
        h = bbox[3] - bbox[1]
        if w <= max_width and h <= max_height:
            return font
        size -= 1
    return ImageFont.truetype(str(font_path), 4)


def generate_stamp(
    name: str,
    date_str: str,
    affiliation: str,
    diameter_mm: float,
    line_width_pt: float,
    font_path: Path,
) -> Image.Image:
    """円形3分割スタンプを RGBA PIL Image で返す。"""
    size = _mm_to_px(diameter_mm)
    img = Image.new("RGBA", (size, size), (0, 0, 0, 0))
    draw = ImageDraw.Draw(img)

    line_w = max(1, int(round(line_width_pt * PT_TO_PX)))
    pad = line_w  # 外周線が切れないよう余白

    # 外周円
    draw.ellipse([pad, pad, size - pad - 1, size - pad - 1],
                 outline=VERMILLION, width=line_w)

    # 3分割の水平区切り線（円の上から 1/3, 2/3 の位置）
    y1 = int(size * 1 / 3)
    y2 = int(size * 2 / 3)
    # 円との交点を計算（円方程式から x 範囲を求める）
    r = (size - 2 * pad) / 2
    cx, cy = size / 2, size / 2
    for y in (y1, y2):
        dy = y - cy
        if abs(dy) >= r:
            continue
        dx = (r * r - dy * dy) ** 0.5
        draw.line([(cx - dx, y), (cx + dx, y)], fill=VERMILLION, width=line_w)

    # 各段の中央にテキスト配置
    # 各段の上端・下端・最大幅
    band_h = size // 3
    bands = [
        (name, 0, y1),
        (date_str, y1, y2),
        (affiliation, y2, size),
    ]
    for text, top, bottom in bands:
        if not text:
            continue
        # その y 位置での円内幅を求める（中央線で計算）
        y_mid = (top + bottom) / 2
        dy = y_mid - cy
        if abs(dy) >= r:
            continue
        chord = 2 * ((r * r - dy * dy) ** 0.5)
        max_w = int(chord * 0.85)  # 余白
        max_h = int((bottom - top) * 0.7)
        font = _fit_font(text, max_w, max_h, font_path)
        bbox = font.getbbox(text)
        tw = bbox[2] - bbox[0]
        th = bbox[3] - bbox[1]
        tx = (size - tw) / 2 - bbox[0]
        ty = (top + bottom - th) / 2 - bbox[1]
        draw.text((tx, ty), text, fill=VERMILLION, font=font)

    return img
```

```python
"""マーク検出用テンプレート画像を assets/marks/ に出力する開発スクリプト。

5種類のマーク（◴◰⊗⊠☆）をシステムフォントで描画し、グレースケールPNGとして保存。
"""
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

MARKS = {
    "mark_arc.png": "◴",
    "mark_square_left.png": "◰",
    "mark_circle_x.png": "⊗",
    "mark_square_x.png": "⊠",
    "mark_star.png": "☆",
}
OUT = Path("assets/marks")
SIZE = 64

def main():
    OUT.mkdir(parents=True, exist_ok=True)
    # Windows標準のフォントでマーク文字を描画
    font_candidates = [
        "C:/Windows/Fonts/seguisym.ttf",  # Segoe UI Symbol
        "C:/Windows/Fonts/YuGothM.ttc",
        "C:/Windows/Fonts/meiryo.ttc",
    ]
    font = None
    for fp in font_candidates:
        if Path(fp).exists():
            font = ImageFont.truetype(fp, 48)
            break
    if font is None:
        raise RuntimeError("シンボル描画可能なフォントが見つかりません")

    for filename, ch in MARKS.items():
        img = Image.new("L", (SIZE, SIZE), 255)
        draw = ImageDraw.Draw(img)
        bbox = font.getbbox(ch)
        w = bbox[2] - bbox[0]
        h = bbox[3] - bbox[1]
        x = (SIZE - w) / 2 - bbox[0]
        y = (SIZE - h) / 2 - bbox[1]
        draw.text((x, y), ch, fill=0, font=font)
        img.save(OUT / filename)
        print(f"wrote {OUT / filename}")

if __name__ == "__main__":
    main()
```

```python
"""PyMuPDFを使ったテキストベースのマーク検出。"""
import fitz


def find_marks_in_text_pdf(doc: fitz.Document, mark_char: str) -> dict[int, list[tuple[float, float, float, float]]]:
    """PDFの各ページから mark_char と一致する文字を検索し、外接矩形のリストを返す。

    返り値: {ページindex: [(x, y, width, height), ...]}
    座標はPDFポイント単位（左上原点ではない、PyMuPDFのrect.x0/y0系）。
    """
    results: dict[int, list[tuple[float, float, float, float]]] = {}
    for i, page in enumerate(doc):
        rects = page.search_for(mark_char)
        results[i] = [(r.x0, r.y0, r.width, r.height) for r in rects]
    return results
```

```python
"""OpenCVテンプレートマッチングによるマーク検出。

スキャンPDFや画像レイヤーPDF向け。PDFを300dpiでレンダリングし、
グレースケールでテンプレートと照合、一致度しきい値以上の位置を返す。
"""
from pathlib import Path
import cv2
import numpy as np
import fitz

RENDER_DPI = 300


def _render_page_gray(page: fitz.Page, dpi: int = RENDER_DPI) -> np.ndarray:
    pix = page.get_pixmap(dpi=dpi)
    img = np.frombuffer(pix.samples, dtype=np.uint8).reshape(pix.height, pix.width, pix.n)
    if pix.n >= 3:
        gray = cv2.cvtColor(img[:, :, :3], cv2.COLOR_RGB2GRAY)
    else:
        gray = img[:, :, 0]
    return gray


def _match_multiscale(gray: np.ndarray, template: np.ndarray, threshold: float) -> list[tuple[int, int, int, int]]:
    """複数スケールでテンプレートマッチング、検出された矩形 (x, y, w, h) を返す。"""
    hits: list[tuple[int, int, int, int, float]] = []
    for scale in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        h = int(template.shape[0] * scale)
        w = int(template.shape[1] * scale)
        if h < 8 or w < 8 or h > gray.shape[0] or w > gray.shape[1]:
            continue
        resized = cv2.resize(template, (w, h), interpolation=cv2.INTER_AREA)
        result = cv2.matchTemplate(gray, resized, cv2.TM_CCOEFF_NORMED)
        ys, xs = np.where(result >= threshold)
        for x, y in zip(xs, ys):
            hits.append((int(x), int(y), w, h, float(result[y, x])))
    return _nms(hits)


def _nms(hits: list[tuple[int, int, int, int, float]], iou_thr: float = 0.3) -> list[tuple[int, int, int, int]]:
    """単純なNon-Maximum Suppression。"""
    if not hits:
        return []
    hits_sorted = sorted(hits, key=lambda h: -h[4])
    kept: list[tuple[int, int, int, int]] = []
    for x, y, w, h, _ in hits_sorted:
        ok = True
        for kx, ky, kw, kh in kept:
            ix1 = max(x, kx); iy1 = max(y, ky)
            ix2 = min(x + w, kx + kw); iy2 = min(y + h, ky + kh)
            iw = max(0, ix2 - ix1); ih = max(0, iy2 - iy1)
            inter = iw * ih
            union = w * h + kw * kh - inter
            if union > 0 and inter / union > iou_thr:
                ok = False
                break
        if ok:
            kept.append((x, y, w, h))
    return kept


def find_marks_in_image_pdf(
    doc: fitz.Document,
    template_path: Path,
    threshold: float = 0.75,
    dpi: int = RENDER_DPI,
) -> dict[int, list[tuple[float, float, float, float]]]:
    """画像PDFからマークを検出。返り値はPDFポイント座標系の (x, y, w, h)。"""
    template = cv2.imread(str(template_path), cv2.IMREAD_GRAYSCALE)
    if template is None:
        raise FileNotFoundError(template_path)

    scale_pdf = 72 / dpi  # ピクセル → PDFポイント
    results: dict[int, list[tuple[float, float, float, float]]] = {}
    for i, page in enumerate(doc):
        gray = _render_page_gray(page, dpi)
        rects = _match_multiscale(gray, template, threshold)
        results[i] = [
            (x * scale_pdf, y * scale_pdf, w * scale_pdf, h * scale_pdf)
            for x, y, w, h in rects
        ]
    return results
```

```python
"""マーク検出ファサード：テキスト検索を試し、見つからなければ画像検出にフォールバック。"""
from pathlib import Path
import fitz
from core.text_finder import find_marks_in_text_pdf
from core.image_finder import find_marks_in_image_pdf

MARK_CHARS = ["◴", "◰", "⊗", "⊠", "☆"]

MARK_TEMPLATE_FILES = {
    "◴": "mark_arc.png",
    "◰": "mark_square_left.png",
    "⊗": "mark_circle_x.png",
    "⊠": "mark_square_x.png",
    "☆": "mark_star.png",
}


def detect_marks(
    doc: fitz.Document,
    mark_char: str,
    template_dir: Path,
) -> dict[int, list[tuple[float, float, float, float]]]:
    """PDFから mark_char の出現位置を検出。

    まずテキスト検索、各ページで0件なら同ページのみ画像検出にフォールバック。
    返り値: {ページindex: [(x, y, w, h), ...]} PDFポイント座標。
    """
    if mark_char not in MARK_CHARS:
        raise ValueError(f"未対応のマーク: {mark_char}")

    text_results = find_marks_in_text_pdf(doc, mark_char)

    needs_image = [i for i, rs in text_results.items() if not rs]
    if needs_image:
        template = template_dir / MARK_TEMPLATE_FILES[mark_char]
        image_results = find_marks_in_image_pdf(doc, template, threshold=0.75)
        for i in needs_image:
            text_results[i] = image_results.get(i, [])
    return text_results
```

```python
"""マーク領域の白塗り＋スタンプ画像挿入＋上書き保存。"""
import io
from pathlib import Path
import fitz
from PIL import Image

MASK_PADDING_PT = 2.0


def stamp_pdf_in_place(
    pdf_path: Path,
    detections: dict[int, list[tuple[float, float, float, float]]],
    stamp_image: Image.Image,
    stamp_diameter_pt: float,
) -> list[int]:
    """検出されたマーク位置に対し、白塗り→スタンプ画像挿入を行い、元PDFに上書き保存。

    detections: {ページindex: [(x, y, w, h), ...]}
    返り値: 押印した1-origin ページ番号のリスト
    """
    doc = fitz.open(pdf_path)
    stamped_pages: list[int] = []

    # スタンプ画像をPNGバイト列に変換（PyMuPDFに渡すため）
    buf = io.BytesIO()
    stamp_image.save(buf, format="PNG")
    stamp_bytes = buf.getvalue()

    half = stamp_diameter_pt / 2

    for page_idx, rects in detections.items():
        if not rects:
            continue
        page = doc[page_idx]
        for x, y, w, h in rects:
            # マーク中心
            cx = x + w / 2
            cy = y + h / 2
            # 白塗り矩形（マーク領域＋余白）
            mask_rect = fitz.Rect(
                x - MASK_PADDING_PT,
                y - MASK_PADDING_PT,
                x + w + MASK_PADDING_PT,
                y + h + MASK_PADDING_PT,
            )
            page.draw_rect(mask_rect, color=(1, 1, 1), fill=(1, 1, 1), overlay=True)
            # スタンプ配置矩形（中心合わせ）
            stamp_rect = fitz.Rect(cx - half, cy - half, cx + half, cy + half)
            page.insert_image(stamp_rect, stream=stamp_bytes, overlay=True)
        stamped_pages.append(page_idx + 1)

    # 上書き保存（incremental だと壊れることがあるので通常save、別名→置換）
    tmp_path = pdf_path.with_suffix(".tmp.pdf")
    doc.save(tmp_path, garbage=4, deflate=True)
    doc.close()
    pdf_path.unlink()
    tmp_path.rename(pdf_path)

    return stamped_pages
```

```python
"""CSV形式の押印履歴ログ（UTF-8 BOM付き、追記モード）。"""
import csv
from pathlib import Path

LOG_HEADERS = ["timestamp", "pdf_path", "pages", "stamp_name", "stamp_date", "stamp_affiliation"]


def default_log_path() -> Path:
    return Path.home() / "Documents" / "StampApp" / "stamp_log.csv"


def append_log(
    log_path: Path,
    timestamp: str,
    pdf_path: str,
    pages: list[int],
    stamp_name: str,
    stamp_date: str,
    stamp_affiliation: str,
) -> None:
    log_path.parent.mkdir(parents=True, exist_ok=True)
    is_new = not log_path.exists()
    with log_path.open("a", encoding="utf-8-sig", newline="") as f:
        writer = csv.writer(f)
        if is_new:
            writer.writerow(LOG_HEADERS)
        writer.writerow([
            timestamp,
            pdf_path,
            ",".join(str(p) for p in pages),
            stamp_name,
            stamp_date,
            stamp_affiliation,
        ])
```

```python
            ).pack(side="left", padx=4)

        ctk.CTkLabel(self, text="検出結果", anchor="w").pack(fill="x", padx=6, pady=(8, 0))
        self.result_label = ctk.CTkLabel(self, text="（PDF未読込）", anchor="w", justify="left")
        self.result_label.pack(fill="x", padx=6, pady=2)

    def _fire(self):
        self._on_mark_change(self.mark_var.get())

    def selected_mark(self) -> str:
        return self.mark_var.get()

    def set_detection_summary(self, mark: str, detections: dict[int, list]):
        pages = [str(i + 1) for i, rs in detections.items() if rs]
        if not pages:
            self.result_label.configure(text=f"{mark} を検出: 0ページ")
        else:
            self.result_label.configure(
                text=f"{mark} を検出: {len(pages)}ページ\n(P{','.join(pages)})"
            )
```

```python
            messagebox.showwarning("StampApp", "PDFが選択されていません")

                date_str=values["date_str"],

                line_width_pt=values["line_width_pt"],
                font_path=FONT_PATH,
            )
            stamped = stamp_pdf_in_place(
                pdf_path=self._pdf_path,
                detections=self._detections,
                stamp_image=stamp_img,
                stamp_diameter_pt=values["diameter_mm"] * 72 / 25.4,
            )
            append_log(
                default_log_path(),
                timestamp=datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                pdf_path=str(self._pdf_path),
                pages=stamped,
                stamp_name=values["name"],
                stamp_date=values["date_str"],
                stamp_affiliation=values["affiliation"],
```
